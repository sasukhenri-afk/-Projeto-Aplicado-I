
"""Normaliza os quatro CSVs publicados pelo Metrô no Projeto Aplicado I."""

import argparse
import csv
from datetime import date
from pathlib import Path
import re

import pandas as pd


ARQUIVOS = {
    "mensal": "Entrada de Passageiros por Linha  - 2025_8.csv",
    "diaria": "Entrada de Passageiros por Estação - Diária - 2025_7.csv",
    "oferta": "Oferta - 2026_4.csv",
    "infraestrutura": "Infraestrutura - 2024_11.csv",
}

MESES = {
    nome.upper(): numero
    for numero, nome in enumerate(
        "Jan Fev Mar Abr Mai Jun Jul Ago Set Out Nov Dez".split(), 1
    )
}

LINHAS = [
    "Linha 1-Azul",
    "Linha 2-Verde",
    "Linha 3-Vermelha",
    "Linha 15-Prata",
]

PADRAO_LINHA = re.compile(r"LINHA\s+(\d+\s*-\s*[A-Z]+)", re.I)
PADRAO_MES = re.compile(r"([A-Z]{3})/(\d{4})", re.I)
NULOS = {"", "-", "–", "—"}


def ler_csv(caminho):
    conteudo = caminho.read_bytes()

    try:
        texto = conteudo.decode("utf-8-sig")
    except UnicodeDecodeError:
        texto = conteudo.decode("cp1252")

    return [
        [valor.strip() for valor in linha]
        for linha in csv.reader(texto.splitlines(), delimiter=";")
    ]


def numero_br(valor):
    if valor in NULOS:
        return None

    if not re.fullmatch(
        r"(?:\d{1,3}(?:\.\d{3})+|\d+)(?:,\d+)?", valor
    ):
        raise ValueError(f"Valor numérico inesperado: {valor!r}")

    return float(valor.replace(".", "").replace(",", "."))


def nome_linha(titulo):
    trecho = PADRAO_LINHA.search(titulo)[1]
    return "Linha " + re.sub(r"\s+", "", trecho).title()


def preparar_mensal(linhas):
    registros = []

    for indice, linha in enumerate(linhas[:-1]):
        for coluna, titulo in enumerate(linha):

            if re.fullmatch(r"REDE(?:\s+\*)?", titulo, re.I):
                grupo = "Rede"

            elif PADRAO_LINHA.match(titulo):
                grupo = nome_linha(titulo)

            else:
                continue

            cabecalho = linhas[indice + 1][coluna:coluna + 6]

            if cabecalho != [
                "Mês", "Total", "MDU", "MSA", "MDO", "MAX"
            ]:
                raise ValueError(
                    f"Cabeçalho mensal inesperado em {titulo}: {cabecalho}"
                )

            for linha_dados in linhas[indice + 2:]:
                mes = MESES.get(linha_dados[coluna].upper())

                if mes is None:
                    break

                total, mdu, msa, mdo, maximo = map(
                    numero_br,
                    linha_dados[coluna + 1:coluna + 6]
                )

                registros.append(
                    (2025, mes, grupo, total, mdu, msa, mdo, maximo)
                )

    return pd.DataFrame(registros, columns=[
        "ano",
        "mes",
        "linha",
        "entradas_mil",
        "mdu_mil",
        "msa_mil",
        "mdo_mil",
        "max_mil",
    ])


def preparar_diaria(linhas):
    estacoes, totais = [], []

    for indice, linha in enumerate(linhas[:-1]):
        for inicio, titulo in enumerate(linha):

            if not titulo.upper().startswith("ENTRADAS POR ESTAÇÃO"):
                continue

            grupo = nome_linha(titulo)

            mes_texto, ano = PADRAO_MES.search(titulo).groups()
            mes = MESES[mes_texto.upper()]

            cabecalho = linhas[indice + 1]
            fim = cabecalho.index("TOTAL", inicio + 1)

            codigos = cabecalho[inicio + 1:fim]

            if (
                cabecalho[inicio] != "DIA"
                or len(set(codigos)) != len(codigos)
                or "" in codigos
            ):
                raise ValueError(
                    f"Cabeçalho diário inesperado em {titulo}"
                )

            for linha_dados in linhas[indice + 2:]:
                dia = linha_dados[inicio]

                if not dia.isdecimal():
                    break

                data = date(int(ano), mes, int(dia))

                total = numero_br(linha_dados[fim])

                totais.append((data, grupo, total))

                for codigo, valor in zip(
                    codigos,
                    linha_dados[inicio + 1:fim]
                ):
                    estacoes.append(
                        (data, grupo, codigo, numero_br(valor), valor)
                    )

    return (
        pd.DataFrame(
            estacoes,
            columns=[
                "data",
                "linha",
                "estacao",
                "entradas_mil",
                "valor_fonte",
            ],
        ),
        pd.DataFrame(
            totais,
            columns=["data", "linha", "entradas_mil"],
        ),
    )


def preparar_oferta(linhas):
    secoes = {
        "Intervalo mínimo entre trens (segundos)":
            "intervalo_minimo_segundos",

        "Velocidade comercial (Km/h)":
            "velocidade_comercial_kmh",
    }

    registros = []

    for indice, linha in enumerate(linhas[:-1]):
        metrica = secoes.get(linha[0])

        if metrica is None:
            continue

        cabecalho = linhas[indice + 1]

        if cabecalho != ["Mês", *LINHAS]:
            raise ValueError(
                f"Cabeçalho de oferta inesperado: {cabecalho}"
            )

        for linha_dados in linhas[indice + 2:]:
            mes = MESES.get(linha_dados[0].upper())

            if mes is None:
                break

            for grupo, valor in zip(
                cabecalho[1:],
                linha_dados[1:]
            ):
                registros.append(
                    (2026, mes, grupo, metrica, numero_br(valor))
                )

    return pd.DataFrame(
        registros,
        columns=["ano", "mes", "linha", "metrica", "valor"],
    )


def preparar_infraestrutura(linhas):
    colunas = ["linha", "caracteristica", "valor_fonte"]

    inicio = next(
        i for i, linha in enumerate(linhas)
        if linha[0] == "Características"
    )

    grupos = linhas[inicio][1:]

    if grupos != [*LINHAS, "Rede"]:
        raise ValueError(
            f"Cabeçalho de infraestrutura inesperado: {grupos}"
        )

    registros = []

    for linha in linhas[inicio + 1:]:

        if not linha[0] or linha[0].startswith(
            ("¹", "²", "³", "Fonte:")
        ):
            break

        for grupo, valor in zip(grupos, linha[1:]):
            registros.append((grupo, linha[0], valor))

    resultado = pd.DataFrame(registros, columns=colunas)

    resultado.insert(0, "ano", 2024)

    return resultado


def validar_chaves(tabela, chaves, nome):
    if tabela.empty or tabela[chaves].isna().any().any():
        raise ValueError(
            f"{nome}: tabela vazia ou chave ausente"
        )

    duplicadas = tabela.loc[
        tabela.duplicated(chaves, keep=False),
        chaves
    ]

    if not duplicadas.empty:
        raise ValueError(
            f"{nome}: chaves duplicadas\n"
            f"{duplicadas.head().to_string(index=False)}"
        )


def main():
    parser = argparse.ArgumentParser(description=__doc__)

    parser.add_argument(
        "--dados",
        type=Path,
        required=True,
        help="Pasta dados/ do repositório"
    )

    parser.add_argument(
        "--saida",
        type=Path,
        default=Path("dados_tratados")
    )

    args = parser.parse_args()

    entradas = {
        chave: ler_csv(args.dados / arquivo)
        for chave, arquivo in ARQUIVOS.items()
    }

    estacoes, linhas_diarias = preparar_diaria(
        entradas["diaria"]
    )

    tabelas = {
        "demanda_mensal": (
            preparar_mensal(entradas["mensal"]),
            ["ano", "mes", "linha"],
        ),

        "demanda_diaria_estacao": (
            estacoes,
            ["data", "linha", "estacao"],
        ),

        "demanda_diaria_linha": (
            linhas_diarias,
            ["data", "linha"],
        ),

        "oferta_mensal": (
            preparar_oferta(entradas["oferta"]),
            ["ano", "mes", "linha", "metrica"],
        ),

        "infraestrutura": (
            preparar_infraestrutura(entradas["infraestrutura"]),
            ["ano", "linha", "caracteristica"],
        ),
    }

    for nome, (tabela, chaves) in tabelas.items():
        validar_chaves(tabela, chaves, nome)

    args.saida.mkdir(parents=True, exist_ok=True)

    for nome, (tabela, chaves) in tabelas.items():

        tabela.sort_values(chaves).to_csv(
            args.saida / f"{nome}.csv",
            index=False,
            encoding="utf-8-sig",
        )

        print(f"{nome}: {len(tabela):,} registros")


if __name__ == "__main__":
    main()


"""Compara sábados e domingos do mesmo fim de semana usando os totais diários."""

import argparse
from pathlib import Path

import pandas as pd


def parear(tabela, chaves, domingos):
    fim_de_semana = tabela.loc[
        tabela.data.dt.dayofweek.ge(5)
    ].copy()

    fim_de_semana["tipo_dia"] = (
        fim_de_semana.data.dt.dayofweek.map({
            5: "sabado",
            6: "domingo"
        })
    )

    fim_de_semana["data_domingo"] = (
        fim_de_semana.data
        + pd.to_timedelta(
            6 - fim_de_semana.data.dt.dayofweek,
            unit="D"
        )
    )

    combinacoes = tabela[chaves].drop_duplicates().merge(
        pd.DataFrame({"data_domingo": domingos}),
        how="cross"
    )

    indice = [*chaves, "data_domingo"]

    pares = fim_de_semana.pivot(
        index=indice,
        columns="tipo_dia",
        values="entradas_mil"
    )

    pares = pares.reindex(
        index=pd.MultiIndex.from_frame(
            combinacoes[indice]
        ),
        columns=["sabado", "domingo"]
    ).reset_index()

    pares = pares.rename(columns={
        "sabado": "sabado_mil",
        "domingo": "domingo_mil"
    })

    pares.columns.name = None

    pares["data_sabado"] = (
        pares.data_domingo - pd.Timedelta(days=1)
    )

    pares["mes_domingo"] = (
        pares.data_domingo.dt.strftime("%Y-%m")
    )

    pares["par_valido"] = (
        pares[["sabado_mil", "domingo_mil"]]
        .notna()
        .all(axis=1)
    )

    pares["diferenca_mil"] = (
        pares.domingo_mil - pares.sabado_mil
    )

    pares["domingo_menor"] = (
        pares.diferenca_mil.lt(0)
        .astype("boolean")
        .where(pares.par_valido)
    )

    return pares.sort_values(indice)


def resumir(pares, chaves):
    cobertura = pares.groupby(
        chaves,
        as_index=False
    ).agg(
        pares_previstos=("par_valido", "size"),
        pares_validos=("par_valido", "sum"),
    )

    perfil = (
        pares.loc[pares.par_valido]
        .groupby(chaves, as_index=False)
        .agg(
            sabado_media_mil=("sabado_mil", "mean"),
            domingo_media_mil=("domingo_mil", "mean"),
            domingo_menor_qtd=("domingo_menor", "sum"),
            diferenca_mediana_mil=("diferenca_mil", "median"),
        )
    )

    resultado = cobertura.merge(
        perfil,
        on=chaves,
        how="left",
        validate="one_to_one"
    )

    resultado["domingo_menor_qtd"] = (
        resultado.domingo_menor_qtd
        .fillna(0)
        .astype(int)
    )

    resultado["domingo_menor_pct"] = (
        100 * resultado.domingo_menor_qtd
        / resultado.pares_validos.where(
            resultado.pares_validos.gt(0)
        )
    )

    resultado["domingo_vs_sabado_pct"] = (
        100 * (
            resultado.domingo_media_mil
            / resultado.sabado_media_mil.where(
                resultado.sabado_media_mil.gt(0)
            ) - 1
        )
    )

    return resultado


def ler(pasta, nome, chaves):
    tabela = pd.read_csv(
        pasta / f"{nome}.csv",
        encoding="utf-8-sig",
        parse_dates=["data"]
    )

    if (
        tabela.empty
        or tabela[chaves].isna().any().any()
        or tabela.duplicated(chaves).any()
    ):
        raise ValueError(
            f"{nome}: tabela vazia ou chaves incompletas ou duplicadas"
        )

    return tabela


def main():
    parser = argparse.ArgumentParser(
        description=__doc__
    )

    parser.add_argument(
        "--dados",
        type=Path,
        default=Path("dados_tratados")
    )

    parser.add_argument(
        "--saida",
        type=Path,
        default=Path("resultados")
    )

    args = parser.parse_args()

    linhas = ler(
        args.dados,
        "demanda_diaria_linha",
        ["data", "linha"]
    )

    estacoes = ler(
        args.dados,
        "demanda_diaria_estacao",
        ["data", "linha", "estacao"]
    )

    domingos = pd.date_range(
        "2025-01-01",
        "2025-12-31",
        freq="W-SUN"
    )

    pares_linha = parear(
        linhas,
        ["linha"],
        domingos
    )

    pares_estacao = parear(
        estacoes,
        ["linha", "estacao"],
        domingos
    )

    todos_validos = (
        pares_linha.groupby("data_domingo")
        .par_valido.all()
    )

    datas_comuns = todos_validos.index[
        todos_validos
    ]

    mesmas_datas = pares_linha.loc[
        pares_linha.data_domingo.isin(datas_comuns)
    ]

    perfil_linha = pd.concat([
        resumir(
            pares_linha,
            ["linha"]
        ).assign(
            escopo="pares_disponiveis_por_linha"
        ),

        resumir(
            mesmas_datas,
            ["linha"]
        ).assign(
            escopo="mesmas_datas_em_todas_as_linhas"
        ),
    ], ignore_index=True)

    saidas = {
        "pares_por_linha": pares_linha,

        "resumo_por_linha": perfil_linha,

        "resumo_por_linha_mes": resumir(
            pares_linha,
            ["linha", "mes_domingo"]
        ),

        "resumo_por_estacao": resumir(
            pares_estacao,
            ["linha", "estacao"]
        ),
    }

    args.saida.mkdir(
        parents=True,
        exist_ok=True
    )

    for nome, tabela in saidas.items():
        tabela.to_csv(
            args.saida / f"{nome}.csv",
            index=False,
            encoding="utf-8-sig",
            float_format="%.4f"
        )

    print(
        f"{len(domingos)} fins de semana no calendário de 2025."
    )

    print(
        f"{len(datas_comuns)} pares completos em comum entre as linhas."
    )

    print(
        f"{len(saidas)} tabelas em {args.saida}."
    )


if __name__ == "__main__":
    main()
